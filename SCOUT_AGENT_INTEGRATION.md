## 📊 КООРДИНАЦИОННЫЙ ЦЕНТР

### Главный Координатор (Manager Agent)

```python
class ManagerAgent:
    """Координирует всех 7 агентов"""
    
    def __init__(self):
        self.strategist = StrategistAgent()
        self.trend_analyst = TrendAnalystAgent()
        self.scout = ScoutAgent()  # ⭐ НОВЫЙ
        self.copywriter = CopywriterAgent()
        self.format_creator = FormatCreatorAgent()
    
    def execute_task(self, task):
        """
        Выполняет задачу с использованием нужных агентов
        """
        
        if "вирусное видео" in task.lower():
            # 1. Отправить Scout на поиск
            scout_results = self.scout.find_viral_videos(
                topic=task.get("topic"),
                niche=task.get("niche")
            )
            
            # 2. Отправить результаты Стратегу
            strategy = self.strategist.analyze(scout_results)
            
            # 3. Трендолог определяет факторы
            trends = self.trend_analyst.identify_factors(
                scout_results
            )
            
            # 4. Копирайтер создает описания
            copy = self.copywriter.write_descriptions(
                scout_results
            )
            
            # 5. Мультиформатник адаптирует
            formats = self.format_creator.adapt_formats(
                scout_results,
                platforms=["youtube", "tiktok", "instagram"]
            )
            
            # 6. Вернуть результаты
            return {
                "scout_findings": scout_results,
                "strategy": strategy,
                "trends": trends,
                "copy": copy,
                "formats": formats
            }
```
